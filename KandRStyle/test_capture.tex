\documentclass{book}
\usepackage{titlesec}
\usepackage{xpatch}

\titleformat{\chapter}[hang]{\large}{\thechapter}{1em}{}

\newcommand{\currentchaptertitle}{DEFAULT}

\makeatletter

% Patch \chapter BEFORE titlesec processes it
\let\oldchapter\chapter
\renewcommand{\chapter}[1]{%
  \renewcommand{\currentchaptertitle}{#1}%
  \typeout{CHAPTER: Captured title: #1}%
  \oldchapter{#1}%
}

\makeatother

The captured title is: ``\currentchaptertitle''
